import{readFile,writeFile,mkdir}from'node:fs/promises';import{createHash}from'node:crypto';
const repo='/home/kulmisiu/dev/dota2-trivia',dir='/tmp/queue-arcade-deep-review/shopkeeper-icon-cache';await mkdir(dir,{recursive:true});
const raw=await readFile(repo+'/src/data/shopkeeper-catalog.json');const catalog=JSON.parse(raw);const sha=b=>createHash('sha256').update(b).digest('hex');
const entries=catalog.groups.flatMap((group,groupIndex)=>group.items.map(item=>({...item,group:group.label,groupIndex})));let next=0;const results=[];
await Promise.all(Array.from({length:4},async()=>{while(next<entries.length){const index=next++,item=entries[index],path=dir+'/'+item.key+'.png';try{let bytes,contentType;if(item.image.startsWith('/')){bytes=await readFile(repo+'/public'+item.image);contentType='image/png';}else{const response=await fetch(item.image,{signal:AbortSignal.timeout(20000)});if(!response.ok)throw Error('HTTP '+response.status);contentType=response.headers.get('content-type');if(!contentType?.startsWith('image/'))throw Error('Unexpected '+contentType);bytes=Buffer.from(await response.arrayBuffer());}if(bytes.length>1024*1024)throw Error('Image exceeds 1 MB');await writeFile(path,bytes);results.push({index:index+1,...item,sourceType:item.image.startsWith('/')?'local-public-file':'public-steam-cdn',contentType,bytes:bytes.length,sha256:sha(bytes),cacheFilename:item.key+'.png'});}catch(e){results.push({index:index+1,...item,error:String(e)})}}}));results.sort((a,b)=>a.index-b.index);await writeFile(dir+'/manifest.json',JSON.stringify({reviewedAt:'2026-10-06',catalogSha256:sha(raw),patch:catalog.patch,clientBuild:catalog.clientBuild,count:entries.length,entries:results},null,2)+'\n');console.log(JSON.stringify({count:entries.length,errors:results.filter(x=>x.error)},null,2));
