import {readFile,writeFile,mkdir,readdir} from 'node:fs/promises';
import {createHash} from 'node:crypto';
import {spawnSync} from 'node:child_process';
import {SOUNDCHECK_DEFINITIONS,soundcheckClips,soundcheckAudioUrl} from '/home/kulmisiu/dev/dota2-trivia/scripts/soundcheck-audit.mjs';
import {soundcheckCutDuration} from '/home/kulmisiu/dev/dota2-trivia/src/lib/soundcheck.mjs';
const root='/tmp/queue-arcade-deep-review/soundcheck-cache'; await mkdir(root,{recursive:true});
const sha=x=>createHash('sha256').update(x).digest('hex');
const jobs=new Map();
for(const d of SOUNDCHECK_DEFINITIONS)for(const c of soundcheckClips(d)){jobs.set(soundcheckAudioUrl(c),{type:'audio',clip:c}); const url='https://dotabase.dillerm.io/vpk/'+c.eventFile;jobs.set(url,{type:'event',path:c.eventFile})}
const queue=[...jobs.entries()];let next=0;const results=[];
await Promise.all(Array.from({length:4},async()=>{while(next<queue.length){const [url,job]=queue[next++]; const path=root+'/'+sha(url)+'.'+(job.type==='audio'?'wav':'txt');try{let bytes;try{bytes=await readFile(path)}catch{const r=await fetch(url,{signal:AbortSignal.timeout(20000)});if(!r.ok)throw Error('HTTP'+r.status); const declared=Number(r.headers.get('content-length'));if(declared>8*1024*1024)throw Error('Asset exceeds8MB');bytes=Buffer.from(await r.arrayBuffer());if(bytes.length>8*1024*1024)throw Error('Asset exceeds8MB');await writeFile(path,bytes)}const result={url,...job,path,bytes:bytes.length,sha256:sha(bytes)};if(job.type==='event'){const local=await readFile('/tmp/queue-arcade-live6882/events/'+job.path);result.localSha256=sha(local);result.exactLocalMatch=result.sha256===result.localSha256;} else {const decoded=spawnSync('ffmpeg',['-v','error','-i',path,'-ac','1','-ar','16000','-f','f32le','pipe:1'],{maxBuffer:4*1024*1024});if(decoded.status!==0)throw Error(decoded.stderr.toString());const b=decoded.stdout;const pcm=new Float32Array(b.buffer,b.byteOffset,b.byteLength/4);const rms=x=>Math.sqrt(x.reduce((sum,v)=>sum+v*v,0)/x.length); result.decodedDuration=pcm.length/16000;result.pcmSha256=sha(b);result.peak=Math.max(...pcm.slice(0,Math.min(pcm.length,100000)).map(Math.abs));result.rms=rms(pcm);result.leadingSilenceMs=pcm.findIndex(x=>Math.abs(x)>0.0001)/16;result.stages=[1,2,3].map(stage=>{const seconds=soundcheckCutDuration(job.clip.duration,stage);const segment=pcm.slice(0,Math.floor(seconds*16000));return{stage,seconds,rms:rms(segment),pcmSha256:sha(Buffer.from(segment.buffer))}});await writeFile(path+'.pcm',b);}results.push(result);}catch(e){results.push({url,...job,error:String(e)})}}}));
await writeFile(root+'/assets.json',JSON.stringify(results,null,2));
console.log(JSON.stringify({assets:results.length,audio:results.filter(x=>x.type==='audio').length,errors:results.filter(x=>x.error),eventMismatches:results.filter(x=>x.type==='event'&&!x.exactLocalMatch).map(x=>x.url),silentFirstCuts:results.filter(x=>x.stages?.[0].rms<0.0001).map(x=>({subject:x.clip.subject,rms:x.stages[0].rms,leading:x.leadingSilenceMs}))},null,2));
