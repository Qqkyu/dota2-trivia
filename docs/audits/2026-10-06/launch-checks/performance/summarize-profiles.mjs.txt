import {readFile,readdir,writeFile} from 'node:fs/promises';import {resolve} from 'node:path';
const dir=resolve(process.argv[2]),files=(await readdir(dir)).filter(f=>f.endsWith('.cpuprofile')),summary=[];
for(const file of files){const p=JSON.parse(await readFile(dir+'/'+file,'utf8')),byId=new Map(p.nodes.map(n=>[n.id,n])),parents=new Map(),exclusive=new Map(),inclusive=new Map();for(const n of p.nodes)for(const child of n.children??[])parents.set(child,n.id);for(let i=0;i<p.samples.length;i++){let id=p.samples[i];exclusive.set(id,(exclusive.get(id)??0)+1);while(id!==undefined){inclusive.set(id,(inclusive.get(id)??0)+1);id=parents.get(id);}}const total=p.samples.length;const rows=(counts)=>[...counts].map(([id,count])=>({samples:count,percent:Math.round(count/total*10000)/100,...byId.get(id).callFrame})).sort((a,b)=>b.samples-a.samples).slice(0,30);summary.push({file,seconds:(p.endTime-p.startTime)/1e6,totalSamples:total,exclusive:rows(exclusive),inclusive:rows(inclusive)});}await writeFile(dir+'/cpu-profile-summary.json',JSON.stringify(summary,null,2)+'\n');console.log(JSON.stringify(summary.map(p=>({file:p.file,seconds:p.seconds,exclusive:p.exclusive.slice(0,12)})),null,2));
