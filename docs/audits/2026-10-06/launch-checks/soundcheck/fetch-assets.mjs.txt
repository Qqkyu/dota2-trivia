import{readFile,writeFile,mkdir,stat,copyFile}from'node:fs/promises';import{createHash}from'node:crypto';
const root='/tmp/queue-arcade-launch-checks/soundcheck',dir=root+'/assets';await mkdir(dir,{recursive:true});const inv=JSON.parse(await readFile(root+'/inventory.json'));const old=JSON.parse(await readFile('/tmp/queue-arcade-deep-review/soundcheck-cache/assets.json'));const prior=new Map(old.map(a=>[a.url,a]));const sha=b=>createHash('sha256').update(b).digest('hex');let next=0,total=0;const rows=[];
await Promise.all(Array.from({length:4},async()=>{while(next<inv.assets.length){const asset=inv.assets[next++],path=dir+'/'+sha(asset.url)+'.wav';try{let bytes,cached=false;try{bytes=await readFile(path);cached=true}catch{const previous=prior.get(asset.url);if(previous&&!previous.error){bytes=await readFile(previous.path);cached=true}else{const r=await fetch(asset.url,{signal:AbortSignal.timeout(20000)});if(!r.ok||!r.headers.get('content-type')?.startsWith('audio/'))throw Error(`HTTP ${r.status} ${r.headers.get('content-type')}`);if(Number(r.headers.get('content-length'))>8*1024*1024)throw Error('Source >8MB');bytes=Buffer.from(await r.arrayBuffer());}if(bytes.length>8*1024*1024)throw Error('Source >8MB');await writeFile(path,bytes);}total+=bytes.length;if(total>512*1024*1024)throw Error('Comparison corpus >512MB');if(bytes.toString('ascii',0,4)!=='RIFF')throw Error('Not a RIFF WAV');rows.push({...asset,file:path,bytes:bytes.length,sha256:sha(bytes),cacheReused:cached});}catch(e){rows.push({...asset,error:String(e)})}if(rows.length%100===0)console.log('Fetched '+rows.length+'/'+inv.assets.length)}}));await writeFile(root+'/assets.json',JSON.stringify(rows,null,2)+'\n');console.log(JSON.stringify({count:rows.length,bytes:total,errors:rows.filter(x=>x.error).map(x=>({path:x.path,error:x.error}))},null,2));
