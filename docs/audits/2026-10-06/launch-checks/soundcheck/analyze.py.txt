def fingerprint(y):
 if len(y)<256:y=np.pad(y,(0,256-len(y)))
 _,_,z=signal.stft(y,fs=8000,nperseg=256,noverlap=192,boundary='zeros');p=np.abs(z)**2

 features=np.stack([np.mean(bands[:,s],axis=1) if len(s) else np.zeros(bands.shape[0]) for s in np.array_split(np.arange(bands.shape[1]),16)],axis=1)
 features=np.log10(features+1e-10);features-=features.mean();features=features.ravel();return features/(np.linalg.norm(features)+1e-12)

def correlation(a,b):
