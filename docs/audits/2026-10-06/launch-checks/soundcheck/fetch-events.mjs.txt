import{readFile,writeFile,mkdir}from'node:fs/promises';import{dirname}from'node:path';import{createHash}from'node:crypto';const root='/tmp/queue-arcade-launch-checks/soundcheck',inv=JSON.parse(await readFile(root+'/inventory.json','utf8')),results=[];let index=0,total=0;await Promise.all(Array.from({length:4},async()=>{while(index<inv.eventFiles.length){const source=inv.eventFiles[index++];const response=await fetch(source.url,{signal:AbortSignal.timeout(30000)});const bytes=Buffer.from(await response.arrayBuffer());if(!response.ok||bytes.length>2_000_000||!bytes.toString('utf8',0,100).includes('kv3'))throw Error('Invalid event source '+source.url);total+=bytes.length;if(total>32_000_000)throw Error('Event source cap');const file=root+'/events/'+source.path;await mkdir(dirname(file),{recursive:true});await writeFile(file,bytes);results.push({url:source.url,path:source.path,bytes:bytes.length,sha256:createHash('sha256').update(bytes).digest('hex'),priorSha256:source.sha256,identicalToEarlierExtract:createHash('sha256').update(bytes).digest('hex')===source.sha256});}}));await writeFile(root+'/event-downloads.json',JSON.stringify(results.sort((a,b)=>a.path.localeCompare(b.path)),null,2)+'\n');console.log(JSON.stringify({files:results.length,bytes:total,unchanged:results.filter(x=>x.identicalToEarlierExtract).length}));
