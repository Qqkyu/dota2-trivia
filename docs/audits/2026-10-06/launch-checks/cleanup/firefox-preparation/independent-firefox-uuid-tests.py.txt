#!/usr/bin/env python3
"""Small fixtures only. Never read or alter real Firefox trees or host processes."""
import importlib.util
import hashlib
import os
from pathlib import Path
import tempfile
import unittest
from unittest.mock import patch

HERE=Path(__file__).resolve().parent
spec=importlib.util.spec_from_file_location('uuid_normalizer',HERE/'normalize-firefox-font-uuid.py')
m=importlib.util.module_from_spec(spec);spec.loader.exec_module(m)
prep,base=m.load_helpers()

class UuidTests(unittest.TestCase):
 def setUp(self):
  self.tmp=tempfile.TemporaryDirectory(prefix='uuid-independent-',dir=HERE);self.addCleanup(self.tmp.cleanup);self.root=Path(self.tmp.name)
  self.source=self.root/'source';self.cache=self.root/'cache';self.backup=self.root/'backup'/'original.bin';self.backup.parent.mkdir(mode=0o700)
  self.old=b'11111111-1111-4111-8111-111111111111';self.new=b'22222222-2222-4222-8222-222222222222'
  for root,data in ((self.source,self.old),(self.cache,self.new)):
   (root/'firefox/fonts').mkdir(parents=True);(root/m.RELATIVE).write_bytes(data)
   for i in range(60):(root/f'file-{i}').write_bytes(b'synthetic')
  (self.source/'firefox/.parentlock').touch()
  for leaf in ('0','downloading'):(self.source/'firefox/updates'/leaf).mkdir(parents=True)
  for module,key,value in [(prep,'SOURCE',self.source),(prep,'DESTINATION',self.cache),(m,'BACKUP',self.backup),(m,'RECEIPT',self.root/'receipt.json'),(m,'SOURCE_HASH',m.sha(self.old)),(m,'CACHE_HASH',m.sha(self.new))]:
   p=patch.object(module,key,value);p.start();self.addCleanup(p.stop)
  rows=base.tree(self.source)
  for rel in prep.RUNTIME:del rows[rel]
  for key,value in [('SOURCE_TREE',base.fingerprint(rows)),('CACHE_TREE',base.fingerprint(base.tree(self.cache)))]:
   p=patch.object(m,key,value);p.start();self.addCleanup(p.stop)

 def test_success_preserves_backup_and_cache_and_runtime(self):
  before=base.tree(self.cache);runtime=prep.runtime_guard(base)
  with patch.object(m,'process_checks') as checks:result=m.normalize(prep,base,True,True,True)
  self.assertEqual(checks.call_count,2);self.assertTrue(result['source_changed']);self.assertFalse(result['cache_changed'])
  self.assertEqual(base.tree(self.cache),before);self.assertEqual((self.source/m.RELATIVE).read_bytes(),self.new)
  self.assertEqual(self.backup.read_bytes(),self.old);self.assertEqual(self.backup.stat().st_mode&0o777,0o600);self.assertEqual(prep.runtime_guard(base),runtime)

 def test_default_is_read_only(self):
  with patch.object(m,'process_checks',side_effect=AssertionError('host probe')):result=m.normalize(prep,base)
  self.assertFalse(result['source_changed']);self.assertFalse(self.backup.exists());self.assertEqual((self.source/m.RELATIVE).read_bytes(),self.old)

 def test_completion_gate_blocks_writes(self):
  with self.assertRaises(RuntimeError):m.normalize(prep,base,True,False,True)
  self.assertFalse(self.backup.exists());self.assertEqual((self.source/m.RELATIVE).read_bytes(),self.old)

 def test_non_uuid_change_refuses(self):
  (self.source/'file-0').write_bytes(b'different')
  with self.assertRaises(RuntimeError):m.check_trees(prep,base)
  self.assertFalse(self.backup.exists())

 def test_existing_backup_collision_preserved(self):
  self.backup.write_bytes(self.new);self.backup.chmod(0o600)
  with self.assertRaises(RuntimeError):m.preserve_original(base,self.old)
  self.assertEqual(self.backup.read_bytes(),self.new)

 def test_source_identity_change_blocks_replace(self):
  _,identity=m.read_uuid(base,self.source/m.RELATIVE,m.SOURCE_HASH)
  source=self.source/m.RELATIVE;source.rename(self.root/'old-marker');source.write_bytes(self.old)
  with self.assertRaises(RuntimeError):m.replace_source(base,self.source,identity,self.new)
  self.assertEqual(source.read_bytes(),self.old)

 def test_symlink_marker_refuses(self):
  source=self.source/m.RELATIVE;source.unlink();source.symlink_to(self.cache/m.RELATIVE)
  with self.assertRaises(base.Refusal):m.read_uuid(base,source,m.CACHE_HASH)

 def test_hardlinked_marker_refuses(self):
  source=self.source/m.RELATIVE;os.link(source,self.root/'other-marker')
  with self.assertRaises(RuntimeError):m.read_uuid(base,source,m.SOURCE_HASH)

if __name__=='__main__':unittest.main(verbosity=2)
