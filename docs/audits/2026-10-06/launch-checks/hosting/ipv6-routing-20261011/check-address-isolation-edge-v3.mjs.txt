import { readFile, lstat, writeFile } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import { request, Agent } from 'node:https';
import { isIP } from 'node:net';
import { execFile } from 'node:child_process';
import { promisify } from 'node:util';
import { pathToFileURL } from 'node:url';
import path from 'node:path';

const BASE = '/tmp/queue-arcade-launch-checks/hosted-smoke-preparation';
const SMOKE = path.join(BASE, 'hosted-smoke.mjs');
const SMOKE_SHA = 'e0a09e4909016e3ee5dbefefafe9d7385401a6972294d10c8d5a92679b7bfc68';
const ORIGIN = 'https://queue-arcade.onrender.com';
const WINDOWS_CURL = '/mnt/c/Windows/System32/curl.exe';
const execute = promisify(execFile);
const digest = (raw) => createHash('sha256').update(raw).digest('hex');
export function ensure(value, code) { if (!value) throw Object.assign(new Error(code), { publicCode: code }); }
export function failure(error) { return /^[A-Z][A-Z0-9_]{1,80}$/.test(error?.publicCode ?? '') ? error.publicCode : 'ISOLATION_CHECK_FAILED'; }
export function validateV2GateBinding(gate) { ensure(gate.ingressProofVersion === 2 && /^[0-9a-f]{64}$/.test(gate.ingressProofDigest ?? ''), 'GATE_INGRESS_V2'); }
export function ipv6(value) {
  const address = value.trim();
  ensure(isIP(address) === 6 && /^[23][0-9a-f]{3}:/i.test(address) && !address.includes('%'), 'IPV6_GLOBAL_ADDRESS');
  return new URL(`http://[${address}]/`).hostname;
}
export function curlArgs(url, headers = false, seconds = 8) {
  ensure(['https://api64.ipify.org/', 'https://www.cloudflare.com/cdn-cgi/trace', ORIGIN + '/api/auth/steam'].includes(url), 'CURL_DESTINATION');
  ensure(Number.isInteger(seconds) && seconds >= 1 && seconds <= 8, 'CURL_TIMEOUT');
  return ['--disable', '--ipv6', '--proto', '=https', '--silent', '--show-error', '--max-time', String(seconds), '--connect-timeout', String(Math.min(4, seconds)), '--max-filesize', '16384', ...(url === ORIGIN + '/api/auth/steam' ? ['--connect-to', 'queue-arcade.onrender.com:443:www.cloudflare.com:443'] : []), ...(headers ? ['--output', 'NUL', '--dump-header', '-'] : []), '--write-out', '\nSTATUS:%{http_code}', url];
}
export function parseCurl(raw) {
  ensure(Buffer.byteLength(raw) <= 16384, 'CURL_OUTPUT_SIZE');
  const match = /\nSTATUS:([0-9]{3})$/.exec(raw);
  ensure(match, 'CURL_STATUS');
  return { status: Number(match[1]), body: raw.slice(0, match.index) };
}
export async function curl6(url, headers = false, seconds = 8) {
  let result;
  try {
    result = await execute(WINDOWS_CURL, curlArgs(url, headers, seconds), { timeout: (seconds + 2) * 1000, killSignal: 'SIGKILL', maxBuffer: 16384, env: { PATH: '/usr/bin:/bin', SystemRoot: 'C:\\Windows', WINDIR: 'C:\\Windows' }, windowsHide: true });
  } catch { throw Object.assign(new Error('IPV6_TRANSPORT'), { publicCode: 'IPV6_TRANSPORT' }); }
  ensure(!result.stderr, 'IPV6_STDERR');
  return parseCurl(result.stdout);
}
export function loginResponse(status, headers) {
  ensure(status === 302 || status === 429, 'LOGIN_STATUS');
  if (status === 302) {
    let target; try { target = new URL(headers.location); } catch { ensure(false, 'LOGIN_REDIRECT'); }
    ensure(target.origin + target.pathname === 'https://steamcommunity.com/openid/login' && !target.username && !target.password, 'LOGIN_REDIRECT');
    return { status };
  }
  ensure(headers['ratelimit-limit'] === '120' && headers['ratelimit-remaining'] === '0' && /^[0-9]{10}$/.test(headers['ratelimit-reset'] ?? ''), 'LOGIN_RATE_HEADERS');
  return { status, resetAt: Number(headers['ratelimit-reset']) * 1000 };
}
export function parseLoginHeaders(value) {
  // curl prints all response headers. No header values enter the report.
  const blocks = value.trim().split(/\r?\n\r?\n/);
  const block = blocks.at(-1);
  ensure(/^HTTP\/(?:1\.[01]|2|3) (?:302|429)(?:\s|$)/.test(block), 'IPV6_HEADER_STATUS');
  const headers = {};
  for (const line of block.split(/\r?\n/).slice(1)) {
    const match = /^([^:]+):\s*(.*)$/.exec(line);
    ensure(match, 'IPV6_HEADER_FORMAT');
    const name = match[1].toLowerCase();
    if (['location', 'ratelimit-limit', 'ratelimit-remaining', 'ratelimit-reset'].includes(name)) {
      ensure(headers[name] === undefined, 'IPV6_DUPLICATE_HEADER'); headers[name] = match[2];
    }
  }
  return headers;
}
export function probe4(index, agent, deadline) {
  ensure(index === undefined || Number.isInteger(index) && index >= 0 && index < 130, 'PROBE_INDEX');
  ensure(Date.now() + 1000 < deadline, 'RATE_WINDOW');
  const headers = { 'user-agent': 'queue-arcade-address-isolation/1.0' };
  if (index !== undefined) Object.assign(headers, { 'X-Real-IP': `198.51.100.${index + 1}`, 'X-Forwarded-For': `203.0.113.${index + 1}, 192.0.2.${index + 1}` });
  return new Promise((resolve, reject) => {
    let settled = false;
    const finish = (error, value) => { if (settled) return; settled = true; clearTimeout(timer); error ? reject(error) : resolve(value); };
    const outgoing = request(ORIGIN + '/api/auth/steam', { method: 'GET', headers, family: 4, rejectUnauthorized: true, agent }, (response) => {
      let bytes = 0;
      response.on('data', (chunk) => { bytes += chunk.length; if (bytes > 16384) { outgoing.destroy(); finish(Object.assign(new Error('LOGIN_RESPONSE_SIZE'), { publicCode: 'LOGIN_RESPONSE_SIZE' })); } });
      response.on('end', () => { try { finish(null, loginResponse(response.statusCode, response.headers)); } catch (error) { finish(error); } });
      response.on('error', () => finish(Object.assign(new Error('LOGIN_RESPONSE_IO'), { publicCode: 'LOGIN_RESPONSE_IO' })));
    });
    const timer = setTimeout(() => { outgoing.destroy(); finish(Object.assign(new Error('LOGIN_DEADLINE'), { publicCode: 'LOGIN_DEADLINE' })); }, Math.min(8000, deadline - Date.now()));
    outgoing.on('error', () => finish(Object.assign(new Error('LOGIN_REQUEST_IO'), { publicCode: 'LOGIN_REQUEST_IO' })));
    outgoing.end();
  });
}
export async function runIsolation({ source = curl6, probe = probe4, now = Date.now, checkpoint = async () => {} } = {}) {
  const a = await source('https://api64.ipify.org/');
  const b = await source('https://www.cloudflare.com/cdn-cgi/trace');
  ensure(a.status === 200 && b.status === 200, 'IPV6_SOURCE_STATUS');
  const rows = b.body.split(/\r?\n/).filter((row) => row.startsWith('ip='));
  ensure(rows.length === 1 && ipv6(a.body) === ipv6(rows[0].slice(3)), 'IPV6_SOURCES_DISAGREE');
  const report = { version: 3, ipv6ApplicationRoute: 'cloudflare-shared-edge-connect-to', ipv6ConnectionHostname: 'www.cloudflare.com', originHostAndSniPreserved: true, tlsVerificationEnabled: true, spoofHeadersTested: ['x-real-ip', 'x-forwarded-for'], checkedAt: new Date(now()).toISOString(), ok: false, independentIpv6SourcesAgreed: true, deploymentAndConfigurationStable: false, ipv4Requests: 0, ipv6LoginRequests: 0, allowed: 0, blocked: 0, rawAddressesLogged: false, cookiesRetained: false, redirectsFollowed: 0, appDataWrites: 0 };
  const agent = new Agent({ keepAlive: true, maxSockets: 1 });
  const started = now(); const deadline = started + 50_000;
  let resetAt;
  try {
    for (let index = 0; index < 130; index++) {
      ensure(now() + 6000 < deadline, 'RATE_WINDOW');
      report.ipv4Requests++;
      const result = await probe(index, agent, deadline);
      if (result.status === 302) { ensure(report.blocked === 0, 'SPOOF_BYPASS'); report.allowed++; }
      else {
        ensure(result.status === 429 && result.resetAt > now() + 5000, 'RATE_WINDOW');
        ensure(resetAt === undefined || resetAt === result.resetAt, 'RATE_WINDOW_RESET'); resetAt = result.resetAt; report.blocked++;
      }
    }
    ensure(report.allowed <= 120 && report.blocked >= 10, 'SPOOF_NOT_BLOCKED');
    report.ipv4Requests++;
    const plain = await probe(undefined, agent, deadline);
    ensure(plain.status === 429 && plain.resetAt === resetAt, 'PLAIN_SOURCE_NOT_BLOCKED');
    const remaining = Math.floor((Math.min(deadline, resetAt) - now() - 1000) / 1000);
    ensure(remaining >= 1, 'RATE_WINDOW');
    report.ipv6LoginRequests++;
    const control = await source(ORIGIN + '/api/auth/steam', true, Math.min(4, remaining));
    ensure(now() < deadline && now() < resetAt - 500, 'CONTROL_WINDOW');
    ensure(loginResponse(control.status, parseLoginHeaders(control.body)).status === 302, 'INDEPENDENT_SOURCE_BLOCKED');
    report.headerFreeIpv4Blocked = true; report.independentIpv6Allowed = true; report.sameWindowConfirmed = true; report.elapsedMs = now() - started; report.ok = true;
    return report;
  } catch (error) { report.failure = failure(error); throw error; }
  finally { agent.destroy(); await checkpoint(report); }
}
async function main() {
  ensure(process.argv.length === 4 && process.argv[2] === '--gate', 'USAGE');
  const bytes = await readFile(SMOKE); ensure(digest(bytes) === SMOKE_SHA, 'SMOKE_HELPER_CHANGED');
  const { validateGate, validateLocalInputs, Check } = await import(pathToFileURL(SMOKE));
  const info = await lstat(process.argv[3]); ensure(info.isFile() && !info.isSymbolicLink() && (info.mode & 0o077) === 0 && info.size < 16384, 'GATE_FILE');
  const gate = JSON.parse(await readFile(process.argv[3], 'utf8')); validateGate(gate); validateV2GateBinding(gate); await validateLocalInputs(gate);
  const guard = new Check(ORIGIN, gate, false); await guard.deployment();
  const output = path.join(BASE, 'address-isolation-edge-v3-' + Date.now() + '.json');
  const write = async (report) => writeFile(output, JSON.stringify({ ...report, commit: gate.commit, deployId: gate.deployId, ciRunId: gate.ciRunId }, null, 2) + '\n', { mode: 0o600 });
  const report = await runIsolation({ checkpoint: write });
  report.deploymentAndConfigurationStable = false; await write(report);
  try { await guard.deployment(); await validateLocalInputs(gate); report.deploymentAndConfigurationStable = true; }
  catch (error) { report.ok = false; report.failure = failure(error); throw error; }
  finally { await write(report); }
  console.log(JSON.stringify({ ok: true, resultFile: output, ipv4Requests: report.ipv4Requests, ipv6LoginRequests: report.ipv6LoginRequests, allowed: report.allowed, blocked: report.blocked }));
}
if (process.argv[1] && import.meta.url === pathToFileURL(path.resolve(process.argv[1])).href) main().catch((error) => { console.error(JSON.stringify({ ok: false, failure: failure(error) })); process.exitCode = 1; });
