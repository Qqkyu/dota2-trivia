    def assert_no_private(self, value):
        for private in [AUTH_SECRET, API_KEY, DATABASE_URL, ADDRESS]:
            self.assertTrue(private not in value, 'A synthetic private marker escaped into output')

    def attempt_main(self, *, states=None, manifests=None, operators=None, result=None, process=None, addresses=None):
        self.output_path.unlink(missing_ok=True)
        if states is None:
            states = [(copy.deepcopy(self.state), copy.deepcopy(self.private_env)) for _ in range(2)]
        if manifests is None:
            manifests = [copy.deepcopy(self.manifest) for _ in range(2)]
        if operators is None:
            operators = [copy.deepcopy(self.operator) for _ in range(2)]
        if process is None:
            payload = self.result if result is None else result
            process = subprocess.CompletedProcess(['synthetic-cli'], 0, json.dumps(payload).encode(), b'')
        stdout, stderr = io.StringIO(), io.StringIO()
        error = None
        with mock.patch.object(self.module, 'checked_state', side_effect=states), \
             mock.patch.object(self.module, 'checked_local_inputs', side_effect=manifests) as inputs, \
             mock.patch.object(self.module, 'checked_operator_script', side_effect=operators, create=True) as operator, \
             mock.patch.object(self.module, 'ipv4', side_effect=addresses or [ADDRESS, ADDRESS]), \
             mock.patch.object(self.module, 'run_bounded', return_value=process) as run, \
             contextlib.redirect_stdout(stdout), contextlib.redirect_stderr(stderr):
            try:
                self.module.main()
            except Exception as caught:
                error = caught
        self.last_input_checks = inputs
        self.last_operator_checks = operator
        self.assert_no_private(stdout.getvalue() + stderr.getvalue())
        return error, stdout.getvalue(), stderr.getvalue(), run

    def assert_main_rejected(self, **kwargs):
        error, stdout, stderr, run = self.attempt_main(**kwargs)
        self.assertIsInstance(error, RuntimeError)
        self.assertEqual(stdout, '')
        self.assertEqual(stderr, '')
        self.assertFalse(self.output_path.exists(), 'A failed check must not archive proof')
        return run

    def api_fixtures(self):
        return {
            '': {'id': self.module.SID, 'suspended': 'not_suspended', 'serviceDetails': {'plan': 'free'},
                 'autoDeployTrigger': 'off', 'branch': self.candidate['branch']},
            '/deploys?limit=5': [{'deploy': {'id': self.state['deployId'], 'status': 'live',
                                          'commit': {'id': self.candidate['commit']}}}],
            '/env-vars?limit=100': [{'envVar': {'key': key, 'value': value}}
                                  for key, value in self.private_env.items()],
        }

    def test_success_archives_only_reviewed_fields_and_restricts_child_environment(self):
        error, stdout, stderr, run = self.attempt_main()
        self.assertIsNone(error)
        self.assertEqual(stderr, '')
        archived_text = self.output_path.read_text()
        self.assert_no_private(archived_text)
        output = json.loads(stdout)
        self.assertEqual(json.loads(archived_text), output)
        self.assertEqual(output['deployId'], self.state['deployId'])
        self.assertEqual(output['localInputs'], self.manifest)
        self.assertEqual(output['proofVersion'], 2)
        self.assertEqual(output['operatorScript'], self.operator)
        self.assertEqual(run.call_args.args[0], [self.module.NODE, str(self.operator_path), self.module.ORIGIN])
        self.assertTrue(output['deploymentAndConfigurationStable'])
        passed_env = run.call_args.kwargs['env']
        self.assertEqual(set(passed_env), {'PATH', 'ARCADE_AUTH_SECRET', 'INGRESS_EXPECTED_IPV4'})
        self.assertTrue(passed_env['ARCADE_AUTH_SECRET'] == AUTH_SECRET)
        self.assertEqual(run.call_args.kwargs['limit'], 32768)
        self.assertEqual(run.call_args.kwargs['timeout'], 60)

    def test_both_cf_outcomes_and_each_mixed_combination_succeed(self):
        for rejected in ((), (1,), (4,), (1, 4)):
            with self.subTest(rejected=rejected):
                value = copy.deepcopy(self.result)
                for index in rejected:
                    value['checks'][index].update(outcome='edge_rejected', receivedVerifiedEgress=False,
                                                 httpStatus=403, edgeErrorCode=1000)
                error, stdout, stderr, _ = self.attempt_main(result=value)
                self.assertIsNone(error)
                self.assertEqual(stderr, '')
                result = json.loads(stdout)
                self.assertEqual(result['checks'], value['checks'])
                self.assertEqual(result['operatorScript'], self.operator)

    def test_ordinary_cases_cannot_claim_edge_rejection(self):
        for index in (0, 2, 3):
            with self.subTest(index=index):
                value = copy.deepcopy(self.result)
                value['checks'][index].update(outcome='edge_rejected', receivedVerifiedEgress=False,
                                             httpStatus=403, edgeErrorCode=1000)
                self.assert_main_rejected(result=value)

    def test_edge_rejection_rows_require_exact_fields_types_and_values(self):
        changes = [{'receivedVerifiedEgress': True}, {'suppliedHeaderDidNotControlResult': False},
                   {'httpStatus': 200}, {'httpStatus': 403.0}, {'edgeErrorCode': None},
                   {'edgeErrorCode': 1000.0}, {'edgeErrorCode': 1001}, {'outcome': 'unknown'},
                   {'responseBody': 'error code: 1000'}]
        for index in (1, 4):
            for change in changes:
                with self.subTest(index=index, change=change):
                    value = copy.deepcopy(self.result)
                    value['checks'][index].update(outcome='edge_rejected', receivedVerifiedEgress=False,
                                                 httpStatus=403, edgeErrorCode=1000)
                    value['checks'][index].update(change)
                    self.assert_main_rejected(result=value)

    def test_verified_rows_require_exact_types_status_and_no_edge_code(self):
        for change in ({'httpStatus': 200.0}, {'httpStatus': 403}, {'edgeErrorCode': 1000},
                       {'receivedVerifiedEgress': 1}, {'suppliedHeaderDidNotControlResult': 1},
                       {'outcome': 'unknown'}):
            with self.subTest(change=change):
                value = copy.deepcopy(self.result); value['checks'][0].update(change)
                self.assert_main_rejected(result=value)

    def test_v2_version_and_order_are_required(self):
        variants = []
        for version in (1, 3, '2', 2.0, True, None):
            value = copy.deepcopy(self.result); value['proofVersion'] = version; variants.append(value)
        value = copy.deepcopy(self.result); del value['proofVersion']; variants.append(value)
        value = copy.deepcopy(self.result); value['checks'][1], value['checks'][4] = value['checks'][4], value['checks'][1]; variants.append(value)
        value = copy.deepcopy(self.result); value['checks'][1] = copy.deepcopy(value['checks'][0]); variants.append(value)
        value = copy.deepcopy(self.result); del value['checks'][1]['outcome']; variants.append(value)
        for value in variants:
            with self.subTest(version=value.get('proofVersion')):
                self.assert_main_rejected(result=value)

    def test_operator_script_is_exact_regular_and_bounded(self):

    def test_api_denies_every_redirect_without_a_second_request(self):
        for code in [301, 302, 303, 307, 308]:
            with self.subTest(status=code):
                observed = []
                def build(handler):
                    self.assertIsInstance(handler, self.module.NoRedirect)
                    opener = mock.Mock()
                    def open_request(request, timeout):
                        observed.append(request.full_url)
                        self.assertTrue(request.get_header('Authorization') == 'Bearer ' + API_KEY)
                        return handler.redirect_request(request, None, code, 'Redirect', {}, 'https://redirect.example/')
                    opener.open.side_effect = open_request
                    return opener
                with mock.patch.object(urllib.request, 'build_opener', side_effect=build):
                    with self.assertRaises(urllib.error.HTTPError):
                        self.module.api('')
                self.assertEqual(len(observed), 1)
                self.assertTrue(observed[0].startswith('https://api.render.com/'))

    def test_local_inputs_match_exact_commit_bytes_under_a_minimal_environment(self):
        def git(argv, **kwargs):

            self.assertTrue(argv[3].startswith(self.candidate['commit'] + ':'))
            self.assertNotIn('RENDER_API_KEY', kwargs['env'])
            self.assertEqual(kwargs['env']['GIT_CONFIG_GLOBAL'], os.devnull)
            return subprocess.CompletedProcess(argv, 0, self.local_bytes[argv[3].split(':', 1)[1]], b'')
        with mock.patch.object(self.module, 'run_bounded', side_effect=git):
            self.assertEqual(self.module.checked_local_inputs(self.candidate['commit']), self.manifest)

    def test_local_input_mismatch_symlink_oversize_and_bad_commit_are_rejected(self):
        with mock.patch.object(self.module, 'run_bounded', return_value=subprocess.CompletedProcess([], 0, b'changed', b'')):
            with self.assertRaises(RuntimeError):
                self.module.checked_local_inputs(self.candidate['commit'])
        path = self.root / self.module.LOCAL_INPUTS[0]
        other = self.directory / 'synthetic-link-target'
        other.write_bytes(self.local_bytes[self.module.LOCAL_INPUTS[0]])
        path.unlink()
        path.symlink_to(other)
        with mock.patch.object(self.module, 'run_bounded') as run:
            with self.assertRaises(RuntimeError):
                self.module.checked_local_inputs(self.candidate['commit'])
            run.assert_not_called()
        path.unlink()
        path.write_bytes(b'x' * 65537)
        with mock.patch.object(self.module, 'run_bounded') as run:
            with self.assertRaises(RuntimeError):
                self.module.checked_local_inputs(self.candidate['commit'])
            with self.assertRaises(RuntimeError):
                self.module.checked_local_inputs('not-a-commit')
            run.assert_not_called()

    def test_run_bounded_returns_bytes_and_exit_status(self):
        result = self.module.run_bounded([sys.executable, '-c',
            'import os; os.write(1, b"out"); os.write(2, b"err"); raise SystemExit(7)'],
            env={'PATH': '/usr/bin:/bin'}, timeout=2, limit=64)
        self.assertEqual((result.returncode, result.stdout, result.stderr), (7, b'out', b'err'))

    def test_run_bounded_caps_stdout_stderr_and_combined_output(self):
        for code in ['import os; os.write(1, b"x" * 65)', 'import os; os.write(2, b"x" * 65)',
                     'import os; os.write(1, b"x" * 32); os.write(2, b"x" * 33)']:
            with self.subTest(kind='bounded-output'):
                with self.assertRaisesRegex(RuntimeError, 'output exceeds the bound'):
                    self.module.run_bounded([sys.executable, '-c', code], env={'PATH': '/usr/bin:/bin'}, timeout=2, limit=64)

    def test_run_bounded_has_an_absolute_deadline_despite_trickling_output(self):
        started = time.monotonic()
        code = 'import os, time\nwhile True:\n os.write(1, b".")\n time.sleep(0.02)'
        with self.assertRaisesRegex(RuntimeError, 'exceeded its deadline'):
            self.module.run_bounded([sys.executable, '-c', code], env={'PATH': '/usr/bin:/bin'}, timeout=0.25, limit=8192)
        self.assertLess(time.monotonic() - started, 2)

    def test_main_guard_does_not_print_private_exception_messages(self):
        stdout, stderr = io.StringIO(), io.StringIO()
        marker = '|'.join([AUTH_SECRET, API_KEY, DATABASE_URL, ADDRESS])
        with mock.patch.object(Path, 'read_text', side_effect=RuntimeError(marker)), \
             contextlib.redirect_stdout(stdout), contextlib.redirect_stderr(stderr):
            with self.assertRaises(SystemExit) as stopped:
                runpy.run_path(str(WRAPPER), run_name='__main__')
        self.assertEqual(stopped.exception.code, 1)
        self.assertEqual(stderr.getvalue(), '')
        self.assert_no_private(stdout.getvalue())
        self.assertEqual(json.loads(stdout.getvalue())['errorType'], 'RuntimeError')
