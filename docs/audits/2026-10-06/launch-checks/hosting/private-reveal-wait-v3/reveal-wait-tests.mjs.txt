  if(pathname==='/api/private-rooms'&&options.method==='POST')return response(200,{privateRoom:{code:this.owned.code,isHost:true,players:[{}]},room:{id:`dota2:${gate.contentVersion}:private:${this.owned.code}`}});

   if(this.mode==='delayed-before-snapshot'&&this.roundReads===1)this.clock+=2000;

   const data={serverNow,room:{id:this.owned.roomId},playerIdentity:{kind:'guest',id:this.owned.playerId},round:{...round,id,game:'trivia'},question:{answers:['A','B','C','D'],reveal:round.phase==='reveal'?{correctIndex:0}:null},playerVote:this.voteCount&&round.roundNumber===0?{}:null};
   if(this.mode==='delayed-active'&&this.roundReads===1)this.clock+=6500;
