function browserRoute(method = "GET") {
  const state = { aborted: 0, fulfilled: null };
  return { state, request: () => ({ method: () => method, url: () => origin + "/", headers: () => ({ cookie: "synthetic-cookie", authorization: "synthetic-authorization", accept: "text/html", rsc: "1", "next-router-state-tree": "synthetic-tree", "next-router-prefetch": "1", "next-router-segment-prefetch": "synthetic-segment", "next-url": "/play", "x-nextjs-data": "1", "x-middleware-prefetch": "1" }) }), abort: async () => state.aborted++, fulfill: async (value) => { state.fulfilled = value; } };
}
