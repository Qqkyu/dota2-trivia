export function allowedUrl(raw, origin, provider = false) {
  const url = new URL(raw, origin);
  ensure(!url.username && !url.password && !url.hash, "URL_COMPONENTS");
  if (provider) ensure(url.origin === "https://api.render.com" && (url.pathname === `/v1/services/${SERVICE}` || url.pathname.startsWith(`/v1/services/${SERVICE}/`)), "PROVIDER_URL");
  else ensure(url.origin === origin || (url.protocol === "https:" && !url.port && MEDIA_HOSTS.has(url.hostname)), "REQUEST_ORIGIN");
  return url;
}

export function safeFailure(error) {
  return /^[A-Z][A-Z0-9_]{1,80}$/.test(error?.publicCode ?? "") ? error.publicCode : "UNCLASSIFIED_FAILURE";
}
export async function withDeadline(ms, operation, onTimeout) {
  let timer;
  const deadline = new Promise((_, reject) => {
    timer = setTimeout(() => {
      Promise.resolve(onTimeout()).catch(() => undefined);
      reject(Object.assign(new Error("BROWSER_DEADLINE"), { publicCode: "BROWSER_DEADLINE" }));
    }, ms);
  });
  try { return await Promise.race([operation(), deadline]); }
  finally { clearTimeout(timer); }
}
export async function closeOwnedBrowser(close, kill, isStopped, graceMs = 5000, killMs = 5000) {
  let timer;
  const completed = await Promise.race([
    Promise.resolve().then(close).then(() => true, () => false),
    new Promise((resolve) => { timer = setTimeout(() => resolve(false), graceMs); }),
  ]);
  clearTimeout(timer);
  if (completed && isStopped()) return;
  if (!isStopped()) kill();
  const until = Date.now() + killMs;
  while (!isStopped() && Date.now() < until) await new Promise((resolve) => setTimeout(resolve, 20));
  ensure(isStopped(), "BROWSER_CLEANUP_FAILED");
  ensure(completed, "BROWSER_FORCED_TERMINATION");
}
export async function browserProxyRoute(route, check, blocked) {
  const request = route.request();
  if (!["GET", "HEAD"].includes(request.method())) { blocked(); return route.abort(); }
  const url = allowedUrl(request.url(), check.origin);
  if (url.origin === check.origin && url.pathname.startsWith("/api/auth/")) return route.abort();
  const original = request.headers();
  const headers = Object.fromEntries(["accept", "accept-language", "range", ...FRAMEWORK_HEADERS].filter((key) => typeof original[key] === "string").map((key) => [key, original[key]]));
  const response = await check.request(url, { method: request.method(), headers, browser: true, maxBytes: 3 * 1024 * 1024 });
  ensure(response.status >= 200 && response.status < 300, response.status >= 300 && response.status < 400 ? "BROWSER_REDIRECT" : "BROWSER_HTTP_STATUS");
  const safeHeaders = Object.fromEntries(Object.entries(response.headers).filter(([key, value]) => !/^(set-cookie|set-cookie2|connection|transfer-encoding|content-length)$/i.test(key) && typeof value === "string"));
  return route.fulfill({ status: response.status, headers: safeHeaders, body: response.bytes });
}

    }
    this.output.privateRoomCleanupConfirmed = closed;
    ensure(closed, "PRIVATE_CLEANUP_FAILED");
    await this.recordGuestState({ cleanupConfirmed: true });
    if (this.guestReceiptPath) {
      const completed = path.join(PRIVATE_BASE, `completed-owned-guest-${randomBytes(12).toString("hex")}.json`);
      await rename(this.guestReceiptPath, completed);
      this.guestReceiptPath = completed;
    }
  }
  async recoverOwnedGuest() {
    ensure(!this.rehearsal, "RECOVERY_HOSTED_ONLY");
    const info = await lstat(PENDING_GUEST);
    ensure(info.isFile() && !info.isSymbolicLink() && (info.mode & 0o077) === 0 && info.size < 8192, "OWNED_RECEIPT_FILE");
    const receipt = JSON.parse(await readFile(PENDING_GUEST, "utf8"));
    validateOwnedReceipt(receipt, this.gate, this.origin);
    this.owned = { ...receipt.owned }; this.guestReceipt = receipt; this.guestReceiptPath = PENDING_GUEST;
    await this.cleanupOwnedGuest();
    this.output.checks.recovery = { votesSent: 0, createsSent: 0, startsSent: 0, cleanupConfirmed: true };
  }
  async guest() {
    const code = `LAUNCH${randomBytes(5).toString("hex").toUpperCase()}`;
    this.owned = { code, playerId: randomUUID(), roomId: "" };
    let attempted = false;
    try {
      ensure((await this.json("/api/me")).authenticated === false, "GUEST_ACCOUNT_GUARD");
      await this.recordGuestState({}, true);
      attempted = true;
      const created = await this.json("/api/private-rooms", { method: "POST", body: { action: "create", code, playerId: this.owned.playerId, name: "Launch smoke", games: ["trivia"] } });
      ensure(created.privateRoom?.code === code && created.privateRoom.isHost && created.privateRoom.players?.length === 1 && created.room?.id === `dota2:${this.gate.contentVersion}:private:${code}`, "PRIVATE_CREATE");
      this.owned.roomId = created.room.id;
      await this.recordGuestState({ created: true });
      const started = await this.json("/api/private-rooms/start", { method: "POST", body: { code, playerId: this.owned.playerId } });
      ensure(started.privateRoom.code === code && Number.isFinite(started.countdownMs) && started.countdownMs <= 5000, "PRIVATE_START");
      await this.pause(started.countdownMs + 700);
      const query = `/api/rounds?${new URLSearchParams({ gameId: "dota2", privateCode: code, playerId: this.owned.playerId })}`;
      const active = await this.json(query);
      ensure(active.playerIdentity?.kind === "guest" && active.playerIdentity.id === this.owned.playerId && active.room.id === this.owned.roomId && active.round.phase === "answering" && active.round.game === "trivia" && !active.question.answerMode && active.question.reveal === null && active.question.answers.length === 4 && active.round.answerDeadline - active.serverNow > 5000, "PRIVATE_VOTE_PRECONDITIONS");
      await this.recordGuestState({ voteAttempted: true });
      const vote = await this.json("/api/votes", { method: "POST", body: { gameId: "dota2", privateCode: code, roundId: active.round.id, answerIndex: 0, voterId: this.owned.playerId } });
      ensure(vote.accepted === true && vote.playerIdentity?.kind === "guest" && vote.playerIdentity.id === this.owned.playerId && vote.playerVote, "PRIVATE_VOTE_ACCEPTED");
      let reveal;
      for (let attempt = 0; attempt < 3; attempt++) {
        await this.pause(attempt === 0 ? active.round.answerDeadline - active.serverNow + 1200 : 300);
        reveal = await this.json(query);
        if (reveal.round.id !== active.round.id || reveal.round.phase !== "answering") break;
      }
      this.output.checks.privateRevealTiming = { sameRound: reveal.round.id === active.round.id, phase: reveal.round.phase, answerPresent: Boolean(reveal.question.reveal), playerVotePresent: Boolean(reveal.playerVote) };
      ensure(reveal.round.id === active.round.id && reveal.round.phase === "reveal" && reveal.question.reveal && reveal.playerVote, "PRIVATE_REVEAL");
      this.output.checks.privateGuest = { created: true, started: true, votesSent: 1, accepted: true, revealConfirmed: true, authenticated: false, alphaSessionSent: false, publicCalibrationEligible: false, reportsSent: 0, rawPrivateVoteExpiresUnderNormalRetention: true };
    } finally {
      if (attempted) await this.cleanupOwnedGuest();
    }
  }
  async browser() {
    let browser; let server; let browserRequests = 0; let blockedWrites = 0; let exceeded = false; let pageErrors = 0; let proxyFailure = null;
    const browserStart = Date.now();
    try {
      return await withDeadline(LIMITS.browserMs, async () => {
      const env = Object.fromEntries(["PATH", "DISPLAY", "XDG_RUNTIME_DIR", "LD_LIBRARY_PATH", "LANG"].filter((key) => process.env[key]).map((key) => [key, process.env[key]]));
      server = await chromium.launchServer({ executablePath: process.env.CHROME_BIN ?? "/usr/bin/google-chrome", host: "127.0.0.1", headless: true, env, args: ["--mute-audio", "--disable-background-networking", "--disable-sync"] });
      browser = await chromium.connect(server.wsEndpoint(), { timeout: 15_000 });
      const context = await browser.newContext({ viewport: { width: 320, height: 568 }, serviceWorkers: "block" });
      await context.route("**/*", async (route) => {
        const request = route.request();
        if (++browserRequests > LIMITS.browserRequests || Date.now() - browserStart > LIMITS.browserMs) { exceeded = true; return route.abort(); }
        try {
          return await browserProxyRoute(route, this, () => blockedWrites++);
        } catch (error) { proxyFailure ??= safeFailure(error); await route.abort().catch(() => undefined); }
      });
      const page = await context.newPage(); page.setDefaultTimeout(15_000); page.on("pageerror", () => pageErrors++);
      await page.goto(this.origin, { waitUntil: "domcontentloaded", timeout: 30_000 });
      await page.waitForFunction(() => document.querySelector("main")?.dataset.clientReady === "true");
      await page.getByTestId("start-arcade").click();
      await page.waitForFunction(() => document.querySelector("main")?.dataset.phase === "answering", null, { timeout: 35_000 });
      const views = [];
      for (const size of [{ width: 320, height: 568 }, { width: 1440, height: 1000 }]) {
        await page.setViewportSize(size); await page.waitForTimeout(200);
        const layout = await page.evaluate(() => ({ noHorizontalOverflow: document.documentElement.scrollWidth <= innerWidth + 1, hasPrompt: Boolean(document.querySelector('[data-testid="round-prompt"] h2')), hasTimer: Boolean(document.querySelector('[data-testid="round-timer"]')), hasAnswerControl: Boolean(document.querySelector('[data-testid="arcade-answer"], [data-testid="hero-search-input"], [data-testid="shopkeeper-search-field"], [data-testid="flashback-year-slider"], [data-testid="value-order-card"], [data-testid="flashback-order-card"]')) }));
        ensure(layout.noHorizontalOverflow && layout.hasPrompt && layout.hasTimer && layout.hasAnswerControl, "BROWSER_ROUND_LAYOUT");
        views.push({ ...size, ...layout });
      }
      await page.waitForFunction(() => document.querySelector("main")?.dataset.phase === "reveal", null, { timeout: 22_000 });
      ensure(await page.getByTestId("reveal-panel").count() > 0, "BROWSER_REVEAL");
      ensure(!exceeded && pageErrors === 0 && proxyFailure === null, proxyFailure ?? "BROWSER_RUNTIME");
      this.output.checks.browser = { views, liveAnsweringAndReveal: true, votesSent: 0, mutationRequestsBlocked: blockedWrites, requests: browserRequests, requestCategories: this.browserPathCounts, frameworkHeadersForwarded: [...this.frameworkHeadersSeen].sort(), pageErrors, freshContext: true, allCookiesStripped: true, transport: "bounded Node proxy with browser rendering", redirectsFollowed: 0 };
      }, () => browser?.close());
    } finally {
      if (server) {
        const child = server.process();
        await closeOwnedBrowser(
          () => Promise.all([browser?.close(), server.close()]),
          () => child.kill("SIGKILL"),
          () => child.exitCode !== null || child.signalCode !== null,
        );
      }
    }
  }
  async run({ skipFormats = false, skipBrowser = false } = {}) {
    await this.assertNoPendingGuest();
    await this.deployment(); await this.health(); await this.pages();
    if (!skipFormats) await this.formats();
    await this.guest();
    if (!skipBrowser) await this.browser();
    await this.health(); await this.deployment();
    this.output.ok = true; this.output.finishedAt = new Date().toISOString();
    this.output.httpRequests = this.requests; this.output.httpBytes = this.bytes;
    this.output.cleanupHttpRequests = this.cleanupRequests;
    this.output.browserHttpRequests = this.browserRequests;
    return this.output;
  }
}

export { Check };
async function main() {
  const args = process.argv.slice(2); const rehearsal = ["--rehearse", "--rehearse-all"].includes(args[0]); const recovery = args[0] === "--cleanup-owned";
  ensure(rehearsal ? args.length === 1 : recovery ? args.length === 3 && args[1] === "--gate" : args.length === 2 && args[0] === "--gate", "USAGE");
  const gatePath = recovery ? args[2] : args[1];
  let gate;
  if (rehearsal) {
    const response = await fetch("http://127.0.0.1:3334/api/health", { signal: AbortSignal.timeout(5000), redirect: "error" });
    const health = await response.json(); ensure(health.storage?.backend === "file", "REHEARSAL_ISOLATION");
    gate = { contentVersion: health.content.version };
  } else {
    const info = await lstat(gatePath); ensure(info.isFile() && !info.isSymbolicLink() && (info.mode & 0o077) === 0 && info.size < 16_384, "GATE_FILE");
    gate = JSON.parse(await readFile(gatePath, "utf8")); validateGate(gate, Date.now(), recovery); await validateLocalInputs(gate);
  }
  const check = new Check(rehearsal ? "http://127.0.0.1:3334" : ORIGIN, gate, rehearsal);
  try {
    if (recovery) { await check.deployment(); await check.recoverOwnedGuest(); await check.deployment(); check.output.ok = true; }
    else await check.run({ skipFormats: args[0] === "--rehearse" });
    if (!rehearsal) await validateLocalInputs(gate);
  }
  catch (error) { check.output.ok = false; check.output.failure = safeFailure(error); process.exitCode = 1; }
  finally {
    check.output.httpRequests = check.requests; check.output.httpBytes = check.bytes;
    check.output.cleanupHttpRequests = check.cleanupRequests;
    check.output.browserHttpRequests = check.browserRequests;
    const base = "/tmp/queue-arcade-launch-checks/hosted-smoke-preparation";
    await mkdir(base, { recursive: true, mode: 0o700 });
    await chmod(base, 0o700);
    const name = `${rehearsal ? "rehearsal" : "hosted"}-${Date.now()}.json`;
    await writeFile(path.join(base, name), `${JSON.stringify(check.output, null, 2)}\n`, { mode: 0o600, flag: "wx" });
    console.log(JSON.stringify({ ok: check.output.ok, rehearsal, failure: check.output.failure ?? null, privateRoomCleanupConfirmed: check.output.privateRoomCleanupConfirmed ?? null, resultFile: path.join(base, name) }));
  }
}
if (process.argv[1] && import.meta.url === pathToFileURL(path.resolve(process.argv[1])).href) {
  main().catch((error) => { console.error(JSON.stringify({ ok: false, failure: safeFailure(error) })); process.exitCode = 1; });
}
