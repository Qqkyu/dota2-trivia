function errorKind(error) {
 const message=typeof error?.message==="string"?error.message:"";
 const react=/Minified React error #(\d{1,4})/.exec(message);
 if(react)return "REACT_"+react[1];
 for(const [pattern,label] of [[/Failed to fetch/,"FETCH_FAILED"],[/ChunkLoadError|Loading chunk/,"CHUNK_LOAD"],[/hydration|Hydration/,"HYDRATION"],[/net::ERR_FAILED/,"NETWORK_FAILED"],[/Failed to load resource/,"RESOURCE_FAILED"]])if(pattern.test(message))return label;
 return ["TypeError","ReferenceError","SyntaxError","RangeError","Error"].includes(error?.name)?error.name:"OTHER";
}
