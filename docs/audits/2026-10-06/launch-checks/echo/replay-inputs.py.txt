"""Restore the exact final Echo inputs. Never writes inside this evidence archive."""
import argparse
import hashlib
import json
import subprocess
import tempfile
import urllib.parse
import urllib.request
from pathlib import Path

parser = argparse.ArgumentParser()
parser.add_argument("--output-dir", required=True, type=Path)
parser.add_argument("--source-cache", type=Path)
parser.add_argument("--question-id")
args = parser.parse_args()
archive = Path(__file__).resolve().parent
output = args.output_dir.resolve()
if output == archive or archive in output.parents:
    parser.error("Choose an output directory outside this archive.")
data = json.loads((archive / "inputs-final.json").read_text())
records = [r for r in data["records"] if args.question_id is None or r["id"] == args.question_id]
if not records:
    parser.error("The requested question ID is not in the final reviewed inputs.")
(output / "clips").mkdir(parents=True, exist_ok=True)
(output / "sources").mkdir(exist_ok=True)
sha = lambda data: hashlib.sha256(data).hexdigest()
for record in records:
    url = urllib.parse.urlparse(record["sourceUrl"])
    if url.scheme != "https" or url.hostname != "dotabase.dillerm.io":
        raise ValueError("Unexpected audio source host")
    filename = Path(url.path).name
    cached = args.source_cache / filename if args.source_cache else output / "sources" / filename
    if cached.exists():
        raw = cached.read_bytes()
    else:
        with urllib.request.urlopen(record["sourceUrl"], timeout=30) as response:
            raw = response.read(4 * 1024 * 1024 + 1)
    if len(raw) > 4 * 1024 * 1024 or sha(raw) != record["sourceSha256"]:
        raise ValueError(f"The source bytes differ from the review: {record['id']}")
    (output / "sources" / filename).write_bytes(raw)
    for stage in record["stages"]:
        seconds = stage["requestedSeconds"]
        if seconds is None:
            clip = raw
        else:
            with tempfile.TemporaryDirectory(prefix="echo-replay-") as work:
                source = Path(work) / "in.mp3"
                target = Path(work) / "out.mp3"
                source.write_bytes(raw)
                subprocess.run([
                    "ffmpeg", "-nostdin", "-hide_banner", "-loglevel", "error", "-max_alloc", "16777216",
                    "-y", "-protocol_whitelist", "file,pipe", "-threads", "1", "-filter_threads", "1",
                    "-f", "mp3", "-i", str(source), "-ss", "0", "-t", str(seconds), "-vn",
                    "-map_metadata", "-1", "-threads", "1", "-fs", str(4 * 1024 * 1024), str(target),
                ], check=True, timeout=20)
                clip = target.read_bytes()
        if sha(clip) != stage["sha256"]:
            raise ValueError(f"The encoded stage differs from the review: {record['id']} stage {stage['stage']}. Check the recorded ffmpeg version.")
        (output / "clips" / stage["file"]).write_bytes(clip)
    print(f"Restored and verified {record['id']}", flush=True)
data["records"] = records
(output / "inputs.json").write_text(json.dumps(data, indent=2) + "\n")
print(f"Verified {len(records)} sources and {len(records) * 3} exact gameplay files.")
