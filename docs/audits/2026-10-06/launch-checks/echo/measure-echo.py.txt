"""Measure exact Echo playback inputs without modifying the files."""
import concurrent.futures
import argparse
import hashlib
import json
import math
import os
import re
import subprocess
from pathlib import Path

os.environ["OPENBLAS_NUM_THREADS"] = "1"
os.environ["OMP_NUM_THREADS"] = "1"
import numpy as np
from faster_whisper.audio import decode_audio

parser = argparse.ArgumentParser()
parser.add_argument("--input", default="/tmp/queue-arcade-launch-checks/echo/inputs.json")
parser.add_argument("--output", default="/tmp/queue-arcade-launch-checks/echo/signal-checks.json")
args = parser.parse_args()
root = Path(args.input).resolve().parent
inputs = json.loads(Path(args.input).read_text())
decoded = {}
rows = []
db = lambda value: 20 * math.log10(value) if value > 0 else None
for row in inputs["records"]:
    stages = []
    for stage in row["stages"]:
        file = root / "clips" / stage["file"]
        pcm = decode_audio(str(file), sampling_rate=16000)
        decoded[(row["id"], stage["stage"])] = pcm
        signal = pcm.astype(np.float64)
        absolute = np.abs(signal)
        peak = float(np.max(absolute))
        rms = float(np.sqrt(np.mean(signal * signal)))
        frames = [signal[i:i + 320] for i in range(0, len(signal), 320)]
        frame_rms = [float(np.sqrt(np.mean(frame * frame))) for frame in frames]
        active = [i for i, value in enumerate(frame_rms) if value >= 10 ** (-45 / 20)]
        clipped = absolute >= .999
        edges = np.diff(np.r_[False, clipped, False].astype(int))
        lengths = np.where(edges == -1)[0] - np.where(edges == 1)[0]
        stages.append({
            "stage": stage["stage"], "file": stage["file"], "inputSha256": stage["sha256"],
            "pcm16000Sha256": hashlib.sha256(pcm.astype("<f4").tobytes()).hexdigest(),
            "durationSeconds": len(pcm) / 16000,
            "peakDbfs": db(peak), "rmsDbfs": db(rms),
            "dcOffset": float(np.mean(signal)),
            "fractionAtOrAbove0999": float(np.mean(clipped)),
            "longestRunAtOrAbove0999Samples": int(max(lengths, default=0)),
            "first20msFrameAboveMinus45DbSeconds": active[0] * .02 if active else None,
            "last20msFrameAboveMinus45DbSeconds": min(len(signal) / 16000, (active[-1] + 1) * .02) if active else None,
            "framesAboveMinus45Db": len(active), "totalFrames": len(frames),
            "silent": bool(peak == 0),
        })
    rows.append({"id": row["id"], "hero": row["hero"], "stages": stages})

def loudness(row):
    filename = next(stage["file"] for stage in row["stages"] if stage["stage"] == 3)
    proc = subprocess.run([
        "ffmpeg", "-nostdin", "-hide_banner", "-nostats", "-threads", "1", "-filter_threads", "1",
        "-i", str(root / "clips" / filename), "-af", "loudnorm=I=-23:TP=-2:LRA=7:print_format=json", "-f", "null", "-"
    ], capture_output=True, text=True, check=True, timeout=30)
    found = re.findall(r"\{\s*\"input_i\".*?\}", proc.stderr, re.S)
    if len(found) != 1:
        raise ValueError(f"Loudness output missing for {row['id']}")
    return row["id"], json.loads(found[0])

with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
    loudness_results = dict(pool.map(loudness, inputs["records"]))
for row in rows:
    row["fullLoudness"] = loudness_results[row["id"]]

pairs = []
for stage in [1, 2, 3]:
    for i, left in enumerate(rows):
        a = decoded[(left["id"], stage)].astype(np.float64)
        for right in rows[i + 1:]:
            b = decoded[(right["id"], stage)].astype(np.float64)
            size = min(len(a), len(b))
            denominator = np.linalg.norm(a[:size]) * np.linalg.norm(b[:size])
            cosine = float(np.dot(a[:size], b[:size]) / denominator) if denominator else 0
            if abs(cosine) >= .95:
                pairs.append({"stage": stage, "left": left["id"], "right": right["id"], "alignedPcmCosine": cosine})

summary = {
    "clips": len(rows), "stages": sum(len(row["stages"]) for row in rows),
    "zeroSignalStages": [f"{row['id']}:{stage['stage']}" for row in rows for stage in row["stages"] if stage["silent"]],
    "prefixesBelowMinus60DbRms": [f"{row['id']}:{stage['stage']}" for row in rows for stage in row["stages"][:2] if stage["rmsDbfs"] < -60],
    "fullIntegratedLoudnessRangeLufs": [min(float(row["fullLoudness"]["input_i"]) for row in rows), max(float(row["fullLoudness"]["input_i"]) for row in rows)],
    "fullAbovePointOnePercentClippedSamples": [row["id"] for row in rows if row["stages"][2]["fractionAtOrAbove0999"] > .001],
    "alignedPcmSimilarityAtLeast095": pairs,
    "distinctDecodedPcmHashesByStage": {str(stage): len({row["stages"][stage - 1]["pcm16000Sha256"] for row in rows}) for stage in [1, 2, 3]},
}
Path(args.output).write_text(json.dumps({"summary": summary, "records": rows}, indent=2) + "\n")
print(json.dumps(summary, indent=2))
