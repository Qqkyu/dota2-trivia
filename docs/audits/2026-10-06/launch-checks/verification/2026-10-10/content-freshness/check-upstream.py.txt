#!/usr/bin/env python3
"""Compare the reviewed public Valve responses without changing app data."""
import concurrent.futures
import datetime
import hashlib
import json
import pathlib
import re
import tarfile
import time
import urllib.request

ROOT = pathlib.Path.cwd()
ARCHIVE = ROOT / 'docs/audits/2026-10-06/deep-review/sources/valve-api-2026-10-06.tar.gz'
MANIFEST = ARCHIVE.with_name('valve-api-manifest.json')
OUT = ROOT / 'docs/audits/2026-10-06/launch-checks/verification/2026-10-10/content-freshness'
TEMP = pathlib.Path('/tmp/queue-arcade-content-freshness-20261010')
SHA = lambda b: hashlib.sha256(b).hexdigest()
NOW = lambda: datetime.datetime.now(datetime.timezone.utc).isoformat()
started = NOW()
manifest = json.loads(MANIFEST.read_text())
assert SHA(ARCHIVE.read_bytes()) == manifest['archiveSha256']
entries = {r['path']: r for r in manifest['files']}
baseline = {}
with tarfile.open(ARCHIVE) as source:
    for row in manifest['files']:
        raw = source.extractfile(row['path']).read()
        assert len(raw) == row['bytes'] and SHA(raw) == row['sha256'], row['path']
        baseline[row['path']] = raw

urls = {
    'herolist.json': 'https://www.dota2.com/datafeed/herolist?language=english',
    'itemlist.json': 'https://www.dota2.com/datafeed/itemlist?language=english',
    'abilitylist.json': 'https://www.dota2.com/datafeed/abilitylist?language=english',
    'patchlist.json': 'https://www.dota2.com/datafeed/patchnoteslist?language=english',
    'patches/7.41f.json': 'https://www.dota2.com/datafeed/patchnotes?version=7.41f&language=english',
}
for name in baseline:
    match = re.fullmatch(r'(heroes|items|abilities)/(\d+)\.json', name)
    if match:
        group, ident = match.groups()
        noun = {'heroes': 'hero', 'items': 'item', 'abilities': 'ability'}[group]
        urls[name] = f'https://www.dota2.com/datafeed/{noun}data?language=english&{noun}_id={ident}'
    match = re.fullmatch(r'(ability|item)data-(\d+)\.json', name)
    if match:
        noun, ident = match.groups()
        if len(baseline[name]) > 100:
            urls[name] = f'https://www.dota2.com/datafeed/{noun}data?language=english&{noun}_id={ident}'
assert len(urls) <= 360
assert set(urls).issubset(baseline)

def paths(before, after, path='$'):
    if type(before) is not type(after):
        return [path]
    if isinstance(before, dict):
        result = []
        for key in sorted(set(before) | set(after)):
            if key not in before or key not in after:
                result.append(f'{path}.{key}')
            else:
                result.extend(paths(before[key], after[key], f'{path}.{key}'))
        return result
    if isinstance(before, list):
        if len(before) != len(after):
            return [f'{path}.length']
        return [p for i, pair in enumerate(zip(before, after)) for p in paths(*pair, f'{path}[{i}]')]
    return [] if before == after else [path]

class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        return None

def check(item):
    name, url = item
    old = baseline[name]
    row = {'path': name, 'url': url, 'baselineSha256': SHA(old)}
    for attempt in range(1, 3):
        row['attempts'] = attempt
        try:
            request = urllib.request.Request(url, headers={'User-Agent': 'Queue-Arcade-content-freshness/1.0'})
            with urllib.request.build_opener(NoRedirect()).open(request, timeout=20) as response:
                raw = response.read(2000001)
                assert len(raw) <= 2000000, 'Response exceeds byte limit'
                assert response.status == 200
                row['headers'] = {k.lower(): v for k, v in response.headers.items() if k.lower() in ('date', 'age', 'cache-control', 'etag', 'last-modified')}
            parsed = json.loads(raw)
            changes = paths(json.loads(old), parsed)
            row.update({'retrievedAt': NOW(), 'status': 200, 'bytes': len(raw), 'sha256': SHA(raw), 'rawEqual': raw == old, 'jsonEqual': not changes, 'changedPaths': changes})
            if name == 'patchlist.json':
                row['latestPatch'] = max(parsed['patches'], key=lambda p: p['patch_timestamp'])
                row['patchCount'] = len(parsed['patches'])
            if changes:
                dest = TEMP / 'changed' / name
                dest.parent.mkdir(parents=True, exist_ok=True)
                dest.write_bytes(raw)
            break
        except Exception as error:
            if attempt == 2:
                row.update({'retrievedAt': NOW(), 'error': f'{type(error).__name__}: {error}'})
            else:
                time.sleep(1)
    time.sleep(0.1)
    return row

with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
    rows = sorted(pool.map(check, sorted(urls.items())), key=lambda r: r['path'])
errors = [r['path'] for r in rows if 'error' in r]
changes = [r['path'] for r in rows if r.get('jsonEqual') is False]
result = {'startedAt': started, 'finishedAt': NOW(), 'baselineArchiveSha256': manifest['archiveSha256'], 'baselineManifestSha256': SHA(MANIFEST.read_bytes()), 'requests': len(rows), 'requestConcurrency': 4, 'retriesPerRequest': 1, 'summary': {'failed': errors, 'changedJson': changes, 'identicalBytes': sum(r.get('rawEqual') is True for r in rows), 'identicalJson': sum(r.get('jsonEqual') is True for r in rows)}, 'responses': rows}
OUT.mkdir(parents=True, exist_ok=True)
(OUT / 'upstream-comparison.json').write_text(json.dumps(result, indent=2) + '\n')
print(json.dumps({k: v for k, v in result.items() if k != 'responses'}, indent=2))
raise SystemExit(1 if errors or changes else 0)
