FROM node:22-bookworm-slim AS node
FROM ubuntu:24.04
ARG RUNNER_VERSION=2.337.0
ARG RUNNER_SHA256=70920811a4f8ad4328818682bca5c6469c1c942fab52448868071d0063816613
ENV DEBIAN_FRONTEND=noninteractive \
    RUNNER_TOOL_CACHE=/opt/hostedtoolcache \
    AGENT_TOOLSDIRECTORY=/opt/hostedtoolcache \
    PLAYWRIGHT_BROWSERS_PATH=/opt/playwright \
    XDG_RUNTIME_DIR=/tmp/queue-runner-runtime \
    RUNNER_LABEL=queue-arcade-ci-local
COPY --from=node /usr/local/ /usr/local/
RUN apt-get update && apt-get install --yes --no-install-recommends \
    ca-certificates curl git jq sudo unzip xz-utils zip tar gzip \
    docker.io docker-buildx iptables uidmap \
    ffmpeg pulseaudio pulseaudio-utils \
    libicu74 libssl3t64 libkrb5-3 zlib1g libgcc-s1 libstdc++6 \
    && curl --fail --location --retry 3 https://dl.google.com/linux/direct/google-chrome-stable_current_amd64.deb --output /tmp/chrome.deb \
    && apt-get install --yes --no-install-recommends /tmp/chrome.deb \
    && rm /tmp/chrome.deb && rm -rf /var/lib/apt/lists/*
RUN useradd --create-home --uid 1001 --shell /bin/bash runner \
    && usermod --append --groups docker runner \
    && printf '%s\n' 'runner ALL=(ALL) NOPASSWD:ALL' > /etc/sudoers.d/runner \
    && chmod 0440 /etc/sudoers.d/runner \
    && mkdir -p /opt/runner /opt/hostedtoolcache /opt/playwright /tmp/queue-runner-runtime \
    && chmod 0700 /tmp/queue-runner-runtime \
    && chown -R runner:runner /opt/runner /opt/hostedtoolcache /opt/playwright /tmp/queue-runner-runtime
RUN curl --fail --location --retry 3 \
      "https://github.com/actions/runner/releases/download/v${RUNNER_VERSION}/actions-runner-linux-x64-${RUNNER_VERSION}.tar.gz" \
      --output /tmp/runner.tar.gz \
    && printf '%s  %s\n' "${RUNNER_SHA256}" /tmp/runner.tar.gz | sha256sum --check --strict \
    && tar --extract --gzip --file /tmp/runner.tar.gz --directory /opt/runner \
    && rm /tmp/runner.tar.gz \
    && chown -R runner:runner /opt/runner
COPY entrypoint.sh preflight.sh /usr/local/bin/
RUN chmod 0755 /usr/local/bin/entrypoint.sh /usr/local/bin/preflight.sh
WORKDIR /opt/runner
ENTRYPOINT ["/usr/local/bin/entrypoint.sh"]
CMD ["preflight"]
