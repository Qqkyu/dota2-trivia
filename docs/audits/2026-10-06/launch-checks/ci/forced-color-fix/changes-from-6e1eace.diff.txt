+function luminance(rgb) {
+  const values = rgb.map((value) => {
+    const channel = value / 255;
+    return channel <= 0.04045 ? channel / 12.92 : ((channel + 0.055) / 1.055) ** 2.4;
+  });
+  return values[0] * 0.2126 + values[1] * 0.7152 + values[2] * 0.0722;
+}

+function contrast(first, second) {
+  const values = [luminance(first), luminance(second)].sort((a, b) => b - a);
+  return (values[0] + 0.05) / (values[1] + 0.05);
+}

+async function paintedPixels(buffer) {
+  const { data, info } = await sharp(buffer).removeAlpha().raw().toBuffer({ resolveWithObject: true });
+  const counts = new Map();
+  for (let index = 0; index < data.length; index += info.channels) {
+    const color = [...data.subarray(index, index + 3)];
+    const key = color.join(",");
+    const entry = counts.get(key) ?? { color, count: 0 };
+    entry.count += 1;
+    counts.set(key, entry);
+  }
+  const colors = [...counts.values()].sort((a, b) => b.count - a.count);
+  const background = colors[0].color;
+  return {
+    width: info.width,
+    height: info.height,
+    background,
+    maxContrast: Math.max(...colors.map(({ color }) => contrast(color, background))),
+    pixelsAtLeast3: colors.reduce((sum, entry) => sum + (contrast(entry.color, background) >= 3 ? entry.count : 0), 0),
+    pixelsAtLeast45: colors.reduce(
+      (sum, entry) => sum + (contrast(entry.color, background) >= 4.5 ? entry.count : 0),
+      0,
+    ),
+  };
+}

+async function differentPixels(first, second, area = "all", compareShape = false) {
+  const images = await Promise.all(
+    [first, second].map((buffer) => sharp(buffer).removeAlpha().raw().toBuffer({ resolveWithObject: true })),
+  );
+  if (images[0].info.width !== images[1].info.width || images[0].info.height !== images[1].info.height) {
+    return { changed: null, sameDimensions: false };
+  }
+  const { width, height, channels } = images[0].info;
+  const backgrounds = compareShape
+    ? await Promise.all([first, second].map(async (buffer) => (await paintedPixels(buffer)).background))
+    : null;
+  let changed = 0;
+  for (let y = 0; y < height; y += 1) {
+    for (let x = 0; x < width; x += 1) {
+      if (area === "bottom" && y < height - 3) continue;
+      if (area === "border" && x >= 3 && x < width - 3 && y >= 3 && y < height - 3) continue;
+      const index = (y * width + x) * channels;
+      const changedAtPixel = backgrounds
+        ? contrast([...images[0].data.subarray(index, index + 3)], backgrounds[0]) >= 3 !==
+          contrast([...images[1].data.subarray(index, index + 3)], backgrounds[1]) >= 3
+        : [0, 1, 2].some((offset) => Math.abs(images[0].data[index + offset] - images[1].data[index + offset]) > 10);
+      if (changedAtPixel) {
+        changed += 1;
+      }
+    }
+  }
+  return { changed, sameDimensions: true, compareShape };
+}
