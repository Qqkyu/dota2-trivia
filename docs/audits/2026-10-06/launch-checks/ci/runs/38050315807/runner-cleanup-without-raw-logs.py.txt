#!/usr/bin/env python3
"""Own one disposable runner without host bind mounts or persistent settings."""
import datetime
import io
import json
import os
import pathlib
import re
import subprocess
import sys
import tarfile
import uuid

BASE = pathlib.Path('/tmp/queue-arcade-launch-ci')
EVIDENCE = BASE / 'evidence'
STATE = EVIDENCE / 'active-runner.json'
REPO = 'repos/Qqkyu/dota2-trivia'
WINDOWS_DOCKER = '/mnt/c/Program Files/Docker/Docker/resources/bin/docker.exe'
OWNER_LABEL = 'io.queue-arcade.launch-ci-owner'
os.umask(0o077)

def call(argv, *, data=None, check=True, timeout=120):
    result = subprocess.run(argv, input=data, capture_output=True, timeout=timeout)
    if check and result.returncode:
        # Never echo commands or environment, since registration can carry a token.
        raise RuntimeError(f'{pathlib.Path(argv[0]).name} failed with exit {result.returncode}')
    return result

def docker_binary():
    for name in ('docker', WINDOWS_DOCKER):
        try:
            result = call([name, 'info', '--format', '{{.ServerVersion}}'], check=False, timeout=10)
            if result.returncode == 0:
                return name
        except (OSError, subprocess.TimeoutExpired):
            pass
    raise RuntimeError('The existing Docker Desktop daemon is unavailable')

def json_call(argv):
    return json.loads(call(argv).stdout)

def repository_runners():
    pages = json_call(['gh', 'api', '--paginate', '--slurp', REPO + '/actions/runners?per_page=100'])
    return [runner for page in pages for runner in page['runners']]

def check_state(state):
    name = state['name']
    if not re.fullmatch(r'queue-arcade-ci-[0-9]{8}T[0-9]{6}Z', name):
        raise ValueError('Unexpected runner name')
    if state['volume'] != name + '-data' or not re.fullmatch(r'[0-9a-f]{32}', state['owner']):
        raise ValueError('Unexpected runner ownership record')

def write_state(state):
    temporary = STATE.with_suffix('.tmp')
    temporary.write_text(json.dumps(state, indent=2) + '\n')
    temporary.replace(STATE)

def copy_out(docker, name, source, destination, optional=False):
    result = call([docker, 'cp', f'{name}:{source}', '-'], check=not optional)
    if result.returncode:
        return False
    destination.mkdir(parents=True, exist_ok=True)
    with tarfile.open(fileobj=io.BytesIO(result.stdout)) as archive:
        for member in archive.getmembers():
            path = pathlib.PurePosixPath(member.name)
            if path.is_absolute() or '..' in path.parts or member.issym() or member.islnk():
                raise ValueError('Unsafe diagnostic archive entry')
            target = destination.joinpath(*path.parts)
            if member.isdir():
                target.mkdir(parents=True, exist_ok=True)
            elif member.isfile():
                target.parent.mkdir(parents=True, exist_ok=True)
                target.write_bytes(archive.extractfile(member).read())
    return True

def resource(docker, kind, name):
    # A successful list distinguishes absence from a daemon or transport failure.
    names = call([docker, kind, 'ls', '--all', '--format', '{{.Names}}'] if kind == 'container'
                 else [docker, kind, 'ls', '--format', '{{.Name}}']).stdout.decode().splitlines()
    if name not in names:
        return None
    return json_call([docker, kind, 'inspect', name])[0]

def assert_owned(metadata, kind, state):
    labels = metadata['Config'].get('Labels') if kind == 'container' else metadata.get('Labels')
    if (labels or {}).get(OWNER_LABEL) != state['owner']:
        raise ValueError('Refusing to remove a resource without the exact ownership label')

def start():
    docker = docker_binary()
    if STATE.exists() and not json.loads(STATE.read_text()).get('cleanedAt'):
        raise RuntimeError('The previous audit runner needs explicit cleanup before another start')
    stamp = datetime.datetime.now(datetime.timezone.utc).strftime('%Y%m%dT%H%M%SZ')
    name = 'queue-arcade-ci-' + stamp
    volume = name + '-data'
    if resource(docker, 'container', name) or resource(docker, 'volume', volume) or any(runner['name'] == name for runner in repository_runners()):
        raise RuntimeError('The proposed runner name is already used')
    owner = uuid.uuid4().hex
    state = {'name': name, 'volume': volume, 'owner': owner, 'repository': 'Qqkyu/dota2-trivia',
             'dockerBinary': docker, 'tokenStorage': 'container root-only file; stdin copy; no host token file'}
    write_state(state)
    evidence = EVIDENCE / name
    evidence.mkdir(mode=0o700)
    token = json_call(['gh', 'api', '--method', 'POST', REPO + '/actions/runners/registration-token'])['token']
    if not isinstance(token, str) or not token.strip():
        raise ValueError('Missing registration token')
    call([docker, 'volume', 'create', '--label', f'{OWNER_LABEL}={owner}', volume])
    container_id = call([docker, 'create', '--name', name, '--label', f'{OWNER_LABEL}={owner}',
        '--privileged', '--cgroupns=private', '--cpus', '2', '--memory', '5g', '--memory-swap', '5g', '--shm-size', '1g',
        '--mount', f'type=volume,source={volume},target=/var/lib/docker',
        '--env', f'RUNNER_NAME={name}', '--env', f'RUNNER_LABEL=queue-arcade-ci-local,owner-{owner}',
        'queue-arcade-ci-runner:20261006', 'run']).stdout.decode().strip()
    (evidence / 'runner-container-id.txt').write_text(container_id + '\n')
    stream = io.BytesIO()
    with tarfile.open(fileobj=stream, mode='w') as archive:
        value = (token + '\n').encode()
        info = tarfile.TarInfo('run/runner-registration-token')
        info.size = len(value)
        info.mode = 0o600
        archive.addfile(info, io.BytesIO(value))
        for script in ('entrypoint.sh', 'prepare-cgroup-v2.sh'):
            content = (BASE / 'runner-image' / script).read_bytes()
            info = tarfile.TarInfo('usr/local/bin/' + script)
            info.size = len(content)
            info.mode = 0o755
            archive.addfile(info, io.BytesIO(content))
    call([docker, 'cp', '-', name + ':/'], data=stream.getvalue())
    del token, value
    metadata = resource(docker, 'container', name)
    assert_owned(metadata, 'container', state)
    volume_metadata = resource(docker, 'volume', volume)
    assert_owned(volume_metadata, 'volume', state)
    if volume_metadata.get('Driver') != 'local' or volume_metadata.get('Options'):
        raise ValueError('Unexpected runner volume driver or options')
    host = metadata['HostConfig']
    if (host.get('CgroupnsMode') != 'private' or host.get('NanoCpus') != 2000000000
            or host.get('Memory') != 5368709120 or host.get('MemorySwap') != 5368709120
            or host.get('PidMode') not in ('', None) or host.get('UTSMode') not in ('', None)
            or host.get('IpcMode') not in ('private', '') or host.get('NetworkMode') not in ('default', 'bridge')
            or host.get('Binds') or host.get('VolumesFrom')):
        raise ValueError('Unexpected runner namespace or resource boundary')
    mounts = metadata.get('Mounts', [])
    if (len(mounts) != 1 or mounts[0].get('Type') != 'volume'
            or mounts[0].get('Name') != volume or mounts[0].get('Destination') != '/var/lib/docker'):
        raise ValueError('Unexpected runner mount boundary')
    call([docker, 'start', name])
    print('Started disposable runner ' + name + '.')

def cleanup():
    state = json.loads(STATE.read_text())
    check_state(state)
    name, volume = state['name'], state['volume']
    docker = docker_binary()
    evidence = EVIDENCE / name
    evidence.mkdir(mode=0o700, exist_ok=True)
    metadata = resource(docker, 'container', name)
    if metadata:
        assert_owned(metadata, 'container', state)
        if metadata['State']['Running']:
            raise RuntimeError('Runner is running; wait for its job or coordinate cancellation first')
    volume_metadata = resource(docker, 'volume', volume)
    if volume_metadata:
        assert_owned(volume_metadata, 'volume', state)
    problems = []
    def attempt(label, action):
        try:
            return action()
        except Exception as error:
            # Only constant labels and exception types enter the report.
            problems.append({'step': label, 'errorType': type(error).__name__})
            return None
    # All selected evidence is already copied. Do not read raw runner logs.
    def unregister():
        runners = repository_runners()
        for runner in runners:
            if runner['name'] == name:
                if not any(label['name'] == 'owner-' + state['owner'] for label in runner['labels']):
                    raise ValueError('Refusing to unregister a runner without the exact ownership label')
                call(['gh', 'api', '--method', 'DELETE', REPO + '/actions/runners/' + str(runner['id'])])
    attempt('remove matching repository registration', unregister)
    if metadata:
        attempt('remove owned container', lambda: call([docker, 'rm', name]))
    if volume_metadata:
        attempt('remove owned volume', lambda: call([docker, 'volume', 'rm', volume]))
    remaining_container = attempt('verify container removal', lambda: resource(docker, 'container', name))
    remaining_volume = attempt('verify volume removal', lambda: resource(docker, 'volume', volume))
    def runner_remaining():
        response = repository_runners()
        summary = {'total_count': len(response), 'runners': [
            {key: runner[key] for key in ('id', 'name', 'status')} for runner in response]}
        (evidence / 'runners-after-cleanup.json').write_text(json.dumps(summary, indent=2) + '\n')
        return any(runner['name'] == name for runner in response)
    remaining_registration = attempt('verify repository registration removal', runner_remaining)
    verification_failed = any(x['step'].startswith('verify ') for x in problems)
    cleaned = not verification_failed and not remaining_container and not remaining_volume and remaining_registration is False
    report = {'runner': name, 'resourcesRemoved': cleaned, 'problems': problems}
    (evidence / 'cleanup-result.json').write_text(json.dumps(report, indent=2) + '\n')
    if cleaned:
        state['cleanedAt'] = datetime.datetime.now(datetime.timezone.utc).isoformat()
        write_state(state)
    print(json.dumps(report))
    if not cleaned:
        raise RuntimeError('Runner cleanup is incomplete; retry this command before starting another runner')

if __name__ == '__main__':
    if sys.argv[1:] == ['start']:
        start()
    elif sys.argv[1:] == ['cleanup']:
        cleanup()
    else:
        raise SystemExit('Usage: runner-lifecycle.py start|cleanup')
