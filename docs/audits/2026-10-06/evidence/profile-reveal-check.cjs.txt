const { root } = require('./ts-loader.cjs');
const fs = require('node:fs/promises');
const path = require('node:path');
const os = require('node:os');
const assert = require('node:assert/strict');
(async () => {
  const dir = await fs.mkdtemp(path.join(os.tmpdir(), 'qa-profile-reveal-audit-'));
  process.env.DATABASE_URL = '';
  process.env.ARCADE_DATA_FILE = path.join(dir, 'votes.json');
  process.env.ARCADE_ACCOUNT_DATA_FILE = path.join(dir, 'accounts.json');
  process.env.ARCADE_ALPHA_DATA_FILE = path.join(dir, 'alpha.json');
  process.env.ARCADE_VOTER_SALT = 'audit-private-salt-not-production';
  const account = require(root + '/src/lib/account-store.ts');
  const auth = require(root + '/src/lib/auth-session.ts');
  const { NextRequest } = require(root + '/node_modules/next/server');
  const { GAME_CATALOGS } = require(root + '/src/lib/game-catalogs.ts');
  const { parsePublicRoomSettings } = require(root + '/src/lib/public-rooms.ts');
  const { liveRoundForSettings } = require(root + '/src/lib/private-room-rounds.ts');
  const { isKeyedAnswerQuestion, correctAnswerKey } = require(root + '/src/lib/keyed-answer.ts');
  const votes = require(root + '/src/app/api/votes/route.ts');
  const me = require(root + '/src/app/api/me/route.ts');
  const settings = parsePublicRoomSettings({ gameId: 'dota2' });
  let now = Date.now();
  const initial = liveRoundForSettings(settings, now);
  now = initial.round.startedAt + 1000;
  Date.now = () => now;
  const live = liveRoundForSettings(settings, now);
  const question = GAME_CATALOGS.dota2.questions.find(q => q.id === live.round.questionId);
  const user = await account.upsertSteamUser({ steamId: '76561198000000001', displayName: 'Audit fixture', avatarUrl: 'https://avatars.steamstatic.com/a_full.jpg', profileUrl: 'https://steamcommunity.com/profiles/76561198000000001', authenticatedAt: now });
  const session = await auth.issueAccountSession(user.id);
  const headers = { cookie: 'queue_arcade_session=' + session.token, 'content-type': 'application/json' };
  const body = { ...settings, roundId: live.round.id, voterId: 'audit-local-player', ...(isKeyedAnswerQuestion(question) ? { answerKey: correctAnswerKey(question) } : { answerIndex: question.correctIndex }) };
  const voted = await votes.POST(new NextRequest('http://localhost:3000/api/votes', { method: 'POST', headers, body: JSON.stringify(body) }));
  const votePayload = await voted.json();
  assert.equal(voted.status, 200);
  assert.equal(votePayload.accepted, true);
  const profile = await me.GET(new NextRequest('http://localhost:3000/api/me', { headers }));
  const data = await profile.json();
  assert.equal(data.profile.stats.recentRounds[0].correct, true);
  console.log(JSON.stringify({ phase: live.round.phase, remainingAnswerMs: live.round.answerDeadline - now, voteStatus: voted.status, voteResponse: votePayload, profileStatus: profile.status, profileStats: data.profile.stats }, null, 2));
})();
