const { root } = require('./ts-loader.cjs');
const fs = require('node:fs/promises');
const os = require('node:os');
const path = require('node:path');
const assert = require('node:assert/strict');
(async () => {
  const dir = await fs.mkdtemp(path.join(os.tmpdir(), 'qa-file-store-audit-'));
  process.env.DATABASE_URL = '';
  process.env.ARCADE_DATA_FILE = path.join(dir, 'votes.json');
  process.env.ARCADE_ACCOUNT_DATA_FILE = path.join(dir, 'accounts.json');
  process.env.ARCADE_ALPHA_DATA_FILE = path.join(dir, 'alpha.json');
  process.env.ARCADE_VOTER_SALT = 'audit-private-salt-not-production';
  const vote = require(root + '/src/lib/vote-store.ts');
  const account = require(root + '/src/lib/account-store.ts');
  const now = Date.now();
  const input = { roundId: 'audit-round', roomId: 'audit-room', gameId: 'dota2', contentVersion: 'audit', questionId: 'audit-question', questionRevision: 'audit-revision', correctIndex: 0, answerIndex: 0, basePoints: 200, voterId: 'audit-player', submittedAt: now };
  console.log('health-before-write', await vote.getVoteStoreHealth());
  const tempPath = process.env.ARCADE_DATA_FILE + '.' + process.pid + '.tmp';
  await fs.mkdir(tempPath);
  await assert.rejects(vote.recordRoundVote(input));
  const phantom = await vote.getRoundCrowd(input.roundId, input.roomId, input.voterId, true);
  assert.equal(phantom.total, 1);
  console.log('failed-write-visible-in-memory', JSON.stringify(phantom));
  const missingOnDisk = await fs.readFile(process.env.ARCADE_DATA_FILE, 'utf8').then(() => false, e => e.code === 'ENOENT');
  assert.equal(missingOnDisk, true);
  console.log('failed-write-file-missing', missingOnDisk);
  await fs.rmdir(tempPath);
  const retry = await vote.recordRoundVote(input);
  assert.equal(retry.accepted, false);
  console.log('retry-treated-as-duplicate', retry.accepted === false);
  const user = await account.upsertSteamUser({ steamId: '76561198000000001', displayName: 'Audit fixture', avatarUrl: null, profileUrl: 'https://steamcommunity.com/profiles/76561198000000001', authenticatedAt: now });
  await account.createAccountSession({ tokenHash: 'audit-session-hash', userId: user.id, createdAt: now, expiresAt: now + 60000 });
  const accountTemp = process.env.ARCADE_ACCOUNT_DATA_FILE + '.' + process.pid + '.tmp';
  await fs.mkdir(accountTemp);
  await assert.rejects(account.deleteAccount(user.id));
  assert.equal(await account.getUserBySession('audit-session-hash'), null);
  const diskAccounts = JSON.parse(await fs.readFile(process.env.ARCADE_ACCOUNT_DATA_FILE, 'utf8'));
  assert.equal(Object.keys(diskAccounts.users).length, 1);
  console.log('failed-deletion-session-revoked-only-in-memory-and-user-still-on-disk', true);
  console.log('fixtures', dir);
})();
